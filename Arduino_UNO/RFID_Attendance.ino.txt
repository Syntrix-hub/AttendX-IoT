#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define SS_PIN 10
#define RST_PIN 9

MFRC522 mfrc522(SS_PIN, RST_PIN);
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Pins
const int greenLED = 6;
const int redLED = 5;
const int buzzer = 7;

void setup() {

  Serial.begin(9600);

  SPI.begin();
  mfrc522.PCD_Init();

  lcd.init();
  lcd.backlight();

  pinMode(greenLED, OUTPUT);
  pinMode(redLED, OUTPUT);
  pinMode(buzzer, OUTPUT);

  digitalWrite(greenLED, LOW);
  digitalWrite(redLED, LOW);

  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("RFID Attendance");
  lcd.setCursor(0,1);
  lcd.print("Scan Card");

}

void loop() {

  if (!mfrc522.PICC_IsNewCardPresent())
    return;

  if (!mfrc522.PICC_ReadCardSerial())
    return;

  String uid = "";

  for (byte i = 0; i < mfrc522.uid.size; i++) {

    if (mfrc522.uid.uidByte[i] < 0x10)
      uid += "0";

    uid += String(mfrc522.uid.uidByte[i], HEX);
  }

  uid.toUpperCase();

  lcd.clear();

  // =========================
  // MANASA
  // UID : 53317856
  // =========================
  if(uid == "53317856"){

    lcd.setCursor(0,0);
    lcd.print("Welcome");

    lcd.setCursor(0,1);
    lcd.print("Manasa");

    digitalWrite(greenLED, HIGH);

    tone(buzzer,1000);
    delay(300);
    noTone(buzzer);

    // Send to NodeMCU
    Serial.println(uid + ",Manasa");

    delay(2000);

    digitalWrite(greenLED,LOW);

  }

  // =========================
  // SAGARI
  // UID : 8D974E05
  // =========================
  else if(uid == "8D974E05"){

    lcd.setCursor(0,0);
    lcd.print("Welcome");

    lcd.setCursor(0,1);
    lcd.print("Sagari");

    digitalWrite(greenLED,HIGH);

    tone(buzzer,1000);
    delay(300);
    noTone(buzzer);

    // Send to NodeMCU
    Serial.println(uid + ",Sagari");

    delay(2000);

    digitalWrite(greenLED,LOW);

  }

  // =========================
  // INVALID CARD
  // =========================
  else{

    lcd.setCursor(0,0);
    lcd.print("Invalid Card");

    lcd.setCursor(0,1);
    lcd.print("Access Denied");

    digitalWrite(redLED,HIGH);

    for(int i=0;i<3;i++){

      tone(buzzer,500);
      delay(150);
      noTone(buzzer);
      delay(150);

    }

    delay(2000);

    digitalWrite(redLED,LOW);

  }

  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("RFID Attendance");
  lcd.setCursor(0,1);
  lcd.print("Scan Card");

  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();

  delay(500);

}
